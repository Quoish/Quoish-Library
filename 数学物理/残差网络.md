# 深度增加为何可能更难训练

更深的网络通常拥有更大的表达空间，但优化不一定更容易。原始 ResNet 工作关注“退化问题”：某些更深普通网络连训练误差也比浅层高，不能仅归因于过拟合。即便较深模型理论上可通过恒等映射包含浅层解，优化器也未必容易找到它。

# 残差块的表达

残差块将输入沿捷径相加：

$$
y=x+F(x;\theta).
$$

若期望映射为$H(x)$，残差支路学习$F(x)=H(x)-x$。当需要接近恒等映射时，让残差接近零便能实现。这里的“残差”是相对于输入的映射增量，不是监督回归中的标签预测误差。

例如标量映射希望从$x$得到$1.1x$，残差只需表示$0.1x$。这说明参数化方式改变了优化问题的组织，但不意味着所有训练情况下残差数值都很小。

# 梯度为何有额外通路

令$g=\nabla_yL$为列梯度，则

$$
\nabla_xL=(I+J_F(x))^Tg.
$$

恒等路径提供了直接传递梯度的项。多块叠加可以形成较短的信息与梯度路径，通常有利于深层优化；$J_F$仍可能引起抵消、放大或其他不稳定，不能据此保证梯度永不消失。

# 形状不同时怎么办

逐元素相加要求形状匹配。若空间尺寸或通道数改变，使用投影捷径$P(x)$：

$$
y=P(x)+F(x).
$$

常用1×1卷积调整通道，步长2同时下采样。投影路径的导数也不再是恒等矩阵。本例由$(B,8,16,16)$变为$(B,16,8,8)$时，两条支路都执行对应变换。

# BasicBlock、瓶颈与归一化位置

BasicBlock 常使用两个3×3卷积。瓶颈块通过1×1降维、3×3处理、1×1升维控制计算成本。原始后激活形式在相加后使用激活；预激活变体把归一化和激活移入支路之前，以改善恒等路径。

残差连接是一种可复用结构，CNN、Transformer 和扩散模型都可使用。完整 ResNet 还包括具体层数、下采样安排、归一化与训练配方；包含一次相加的网络不等同于标准 ResNet-18。

# 实验应怎样解释

比较残差与普通网络时，应控制数据、深度和优化设置。小任务与浅网络可能看不到明显优势；模型参数量、投影和归一化也会影响结果。本例用于检验形状匹配、梯度和训练闭环，不据单次结果宣称残差网络普遍更优。

```python
"""统一处理项目路径、随机种子、CPU线程与计算设备。"""

# 标准库
from pathlib import Path
import argparse
import random

# 第三方库（由统一环境提供）
import numpy as np
import torch

# 1. 路径统一从 runtime.py 所在目录出发，复制项目后无须改绝对路径。
ROOT = Path(__file__).resolve().parent

# outputs/ 存放本项目的图像、指标与模型权重；不存在时自动创建。
OUT = ROOT / "outputs"
OUT.mkdir(exist_ok=True)


def setup():
    """初始化运行环境并返回 torch.device。

    命令行参数只有 --cpu；ROOT、OUT 是导入时建立的项目路径。
    """

    # 2. 读取命令行选项：--cpu 允许在有显卡的机器上仍强制使用 CPU。
    parser = argparse.ArgumentParser()
    parser.add_argument("--cpu", action="store_true")
    args = parser.parse_args()

    # 3. 固定 Python、NumPy 和 PyTorch 的随机种子；跨设备仍不保证逐位相同。
    random.seed(7)
    np.random.seed(7)
    torch.manual_seed(7)

    # 小型教学网络限制 CPU 线程数，避免线程调度开销过大。
    torch.set_num_threads(2)

    # 4. 仅当 CUDA 可用且没有指定 --cpu 时使用 GPU；调用方负责移动模型和数据。
    device = torch.device(
        "cuda" if torch.cuda.is_available() and not args.cpu else "cpu"
    )
    print("device:", device)

    # 返回 torch.device；各入口脚本共用相同的设备选择规则。
    return device
```

### `data.py`

```python
"""生成带噪声与位置变化的横条／竖条二分类图像。"""

# 第三方库（由统一环境提供）
import torch


def make_data(n, seed):
    """生成 n 张16×16灰度图像。

    返回 x: (n, 1, 16, 16)，y: (n,)；seed 控制噪声、标签与条纹位置。
    """

    # 1. 使用局部随机生成器；改变数据集种子不会改动模型初始化的全局随机状态。
    generator = torch.Generator().manual_seed(seed)

    # 先生成 (n, 1, 16, 16) 的灰度噪声背景；维度顺序为样本、通道、高、宽。
    x = 0.12 * torch.randn(n, 1, 16, 16, generator=generator)

    # 2. 标签0表示横条，1表示竖条；每个样本另取一个条纹位置。
    y = torch.randint(0, 2, (n,), generator=generator)
    positions = torch.randint(3, 12, (n,), generator=generator)

    # 3. 在各自的背景上叠加2像素宽条纹，位置变化让模型不能只记固定坐标。
    for i in range(n):
        p = positions[i].item()

        # 横条固定行范围、跨越多列；竖条固定列范围、跨越多行。
        if y[i] == 0:
            x[i, 0, p : p + 2, 2:14] += 1
        else:
            x[i, 0, 2:14, p : p + 2] += 1

    # 返回图像 x 与整数标签 y；此处在 CPU 生成，入口脚本再统一移动到目标设备。
    return x, y
```

### `model.py`

```python
"""定义可切换捷径连接的卷积块，并组成一个微型图像分类网络。"""

# 第三方库（由统一环境提供）
import torch
from torch import nn


class Block(nn.Module):
    """两层卷积主支路，可选残差捷径和投影下采样。"""

    def __init__(self, cin, cout, stride=1, residual=True):
        """cin/cout 为输入／输出通道数；stride 控制第一层卷积及投影的步长。"""

        super().__init__()

        # 用同一个开关构造普通版和残差版，主支路的卷积安排保持相同。
        self.residual = residual

        # 1. 主支路：卷积、归一化、ReLU，再接卷积和归一化。
        # 第一层可用 stride=2 下采样；第二层保持得到的空间尺寸。
        self.branch = nn.Sequential(
            nn.Conv2d(cin, cout, 3, stride, 1, bias=False),
            nn.BatchNorm2d(cout),
            nn.ReLU(),
            nn.Conv2d(cout, cout, 3, padding=1, bias=False),
            nn.BatchNorm2d(cout),
        )

        # 2. 仅当残差版的通道或空间尺寸变化时，使用1×1卷积匹配形状。
        # 普通版没有参与计算的捷径；尺寸不变时残差版使用恒等映射。
        self.shortcut = (
            nn.Conv2d(cin, cout, 1, stride, bias=False)
            if residual and (cin != cout or stride != 1)
            else nn.Identity()
        )

    def forward(self, x):
        """输入 (B, cin, H, W)，返回匹配 cout 与 stride 的激活结果。"""

        # 先计算 F(x)；相加之前两条支路必须具有完全相同的张量形状。
        y = self.branch(x)

        # 残差版再加入捷径 P(x)，普通版直接使用主支路输出。
        if self.residual:
            y = y + self.shortcut(x)

        # 本例在相加后激活，对应后激活残差块的安排。
        return torch.relu(y)


class SmallNet(nn.Module):
    """使用三个卷积块的图像分类器；residual 选择是否启用各块捷径。"""

    def __init__(self, residual=True):
        """创建模型组件并登记参数；本方法只定义结构，不执行训练。"""

        super().__init__()

        # 3. 整体网络：提取8通道特征 → 同尺寸块 → 投影下采样到16通道 → 同尺寸块。
        # 最后全局平均池化、展平并输出2类分数。
        self.layers = nn.Sequential(
            nn.Conv2d(1, 8, 3, padding=1),
            nn.ReLU(),
            Block(8, 8, residual=residual),
            Block(8, 16, 2, residual),
            Block(16, 16, residual=residual),
            nn.AdaptiveAvgPool2d(1),
            nn.Flatten(),
            nn.Linear(16, 2),
        )

    def forward(self, x):
        """按模型定义传递输入，返回供损失计算或评价使用的结果。"""

        # 逐层执行已登记的组件，最终输出 (B, 2) logits。
        return self.layers(x)
```

### `main.py`

```python
"""先检验投影尺寸和梯度，再比较普通版与残差版的训练结果。"""

# 标准库
import json

# 第三方库（由统一环境提供）
import torch
from torch import nn

# 本项目模块
from runtime import setup, OUT
from data import make_data
from model import Block, SmallNet


def main():
    """先检验投影尺寸和梯度，再比较普通版与残差版的训练结果。

    按分段注释依次准备数据、执行计算并保存结果。
    """

    # 1. 初始化运行环境。
    device = setup()

    # 2. 创建需要输入梯度的 (2, 8, 16, 16) 张量，单独检查投影残差块。
    dummy = torch.randn(2, 8, 16, 16, device=device, requires_grad=True)

    # 通道8→16、步长2；两条支路都应变为 (2, 16, 8, 8)。
    changed = Block(8, 16, 2).to(device)(dummy)

    # 以均值作为标量目标，确认梯度能从块输出回传至输入。
    changed.mean().backward()
    print(
        "projection shape:",
        tuple(changed.shape),
        "input grad norm:",
        dummy.grad.norm().item(),
    )

    # 尺寸是残差相加的必要条件；用断言使配置错误及时暴露。
    assert changed.shape == (2, 16, 8, 8)

    # 3. 生成独立训练／测试图片并移动到设备，避免每个模型使用不同数据。
    x, y = [t.to(device) for t in make_data(256, 7)]
    tx, ty = [t.to(device) for t in make_data(128, 8)]
    report = {}

    # 4. 分别建立普通网络和残差网络；各自从同一初始随机种子开始。
    for residual in [False, True]:
        # 固定种子便于比较；投影结构仍会使两版参数安排有所不同。
        torch.manual_seed(10)
        model = SmallNet(residual).to(device)
        optimizer = torch.optim.Adam(model.parameters(), lr=0.005)

        # 5. 每个模型训练30轮，每次更新使用64个随机打乱的样本。
        for epoch in range(30):
            # 训练时 BatchNorm 使用当前批次统计；后面的 eval 会切换为运行统计。
            model.train()
            for ids in torch.randperm(len(x), device=device).split(64):
                # 清除上一轮梯度；PyTorch 的 backward 默认会累加到 .grad。
                optimizer.zero_grad()

                # 构造当前批次的标量训练目标，反向传播将从这里开始。
                loss = nn.functional.cross_entropy(model(x[ids]), y[ids])

                # 沿计算图计算参数梯度；此时参数数值尚未更新。
                loss.backward()

                # 优化器根据当前梯度真正更新可训练参数。
                optimizer.step()

        # 切换到评价模式；若含 Dropout／BatchNorm，其行为随之切换。
        model.eval()

        # 不记录这一段的求导图；适用于评价、参数初始化或手动更新。
        with torch.no_grad():
            # 6. 仅在独立测试集计算准确率，不再反向传播。
            accuracy = (model(tx).argmax(1) == ty).float().mean().item()

        # 7. 使用不同文件名保存两版权重，并汇总到同一指标文件。
        name = "residual" if residual else "plain"
        report[name] = {
            "test_accuracy": accuracy,
            "last_batch_loss": loss.item(),
        }
        torch.save(model.state_dict(), OUT / (name + ".pt"))
    print(report)
    (OUT / "metrics.json").write_text(json.dumps(report, indent=2), encoding="utf-8")


if __name__ == "__main__":
    main()
```
